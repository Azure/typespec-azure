# A sub-client inherits the client-level API version override for path parameters

## TypeSpec

```tsp
@service
@versioned(Versions)
@server(
  "{endpoint}",
  "Test endpoint",
  {
    endpoint: url,
  }
)
namespace Versioned;

enum Versions {
  v2022_12_01_preview: "2022-12-01-preview",
}

@head
@route("/with-query-api-version")
op withQueryApiVersion(@query("api-version") apiVersion: string): void;

@route("/sub")
interface SubGroup {
  @head
  @route("/with-path-api-version/{apiVersion}")
  withPathApiVersion(@path apiVersion: string): void;
}
```
