```tsp
@armProviderNamespace
namespace Microsoft.Test;

model ToolControls {
  enabled?: boolean;
}

@discriminator("kind")
model Pet {
  kind: string;
}

model Cat extends Pet {
  kind: "cat";
}

model Details {
  toolControls?: ToolControls | null;
  label?: string | null;
  items?: ToolControls[] | null;
  byName?: Record<ToolControls> | null;
  pet?: Pet | null;
  timestamp?: utcDateTime | null;
  metadata?: unknown;
  entries: (ToolControls | null)[];
  values: Record<ToolControls | null>;
  omitted?: string | null;
}

model Input {
  value: Details;
}

@discriminated
union RawInput {
  details: Details,
}

interface NullOps {
  @route("/input")
  @patch
  update(@body input: Input): Input;

  @route("/raw-input")
  @patch
  updateRaw(@body input: RawInput): RawInput;

  @route("/null-body")
  @put
  putNullBody(@body input: string | null): void;
}
```
