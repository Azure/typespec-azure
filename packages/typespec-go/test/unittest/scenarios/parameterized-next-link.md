# A parameterized next link reinjects query parameters

## TypeSpec

```tsp
@service(#{ title: "Widgets" })
namespace Widgets;

model IncludePendingOptions {
  @query
  includePending?: boolean;
}

model Widget {
  id: string;
}

model WidgetPage {
  @pageItems
  values: Widget[];

  @nextLink
  nextLink: Azure.Core.Legacy.parameterizedNextLink<[IncludePendingOptions.includePending]>;
}

@route("/widgets")
@list
op list(...IncludePendingOptions, @query select: string): WidgetPage;
```
