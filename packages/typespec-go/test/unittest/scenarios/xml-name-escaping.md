# `@Xml.name` values containing quotes, backslashes, backticks, or control characters are escaped in generated Go XML models, marshallers, requests, and responses

## TypeSpec

```tsp
@service(#{ title: "XmlEscaping" })
namespace XmlEscaping;

@Xml.name("part\"number")
model WidgetQuoted {
  @Xml.name("field\"quote")
  name: string;

  @Xml.attribute
  @Xml.name("attr`tick")
  id: string;

  @Xml.name("slash\\line\nbreak\r\ttab")
  timestamp?: utcDateTime;

  @Xml.name("items\"wrap")
  items?: WidgetItem[];
}

@Xml.name("item`entry")
model WidgetItem {
  value: string;
}

@Xml.name("part\"; println(\"element-name-is-data\"); //")
model WidgetStatement {
  name: string;

  @Xml.name("nested\"child")
  child?: WidgetItem;
}

@Xml.name("batch\"list")
model WidgetQuotedList is Array<WidgetQuoted>;

@route("/quoted")
@post
op sendQuoted(
  @header contentType: "application/xml",
  @header accept: "application/xml",
  @body body: WidgetQuoted,
): {
  @header contentType: "application/xml";
  @body body: WidgetStatement;
};

@route("/list")
@post
op sendList(
  @header contentType: "application/xml",
  @header accept: "application/xml",
  @body body: WidgetQuotedList,
): {
  @header contentType: "application/xml";
  @body body: WidgetQuotedList;
};
```
