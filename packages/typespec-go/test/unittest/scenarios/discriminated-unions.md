	if rawMsg == nil || string(rawMsg) == "null" {
		return nil, nil
	}
	var m map[string]any
	if err := json.Unmarshal(rawMsg, &m); err != nil {
		return nil, err
	}
	if m == nil {
		return nil, nil
	}
	var b PetClassification
	switch m["kind"] {
	case "cat":
		b = &Cat{}
	case "dog":
		b = &Dog{}
	default:
		b = &Pet{}
	}
	if err := json.Unmarshal(rawMsg, b); err != nil {
		return nil, err
	}
	return b, nil
}
