# Delimited string and enum array model properties use their encoded wire format

## TypeSpec

```tsp
@service(#{ title: "ArrayEncoding" })
namespace ArrayEncoding;

enum Color {
  Red: "red",
  Blue: "blue",
}

union ExtensibleColor {
  string,
  Red: "red",
  Blue: "blue",
}

model Widget {
  @encode(TypeSpec.ArrayEncoding.commaDelimited)
  names: string[];

  @encode(TypeSpec.ArrayEncoding.pipeDelimited)
  optionalNames?: string[];

  @encode(TypeSpec.ArrayEncoding.spaceDelimited)
  colors: Color[];

  @encode(TypeSpec.ArrayEncoding.newlineDelimited)
  extensibleColors: ExtensibleColor[];
}

@route("/widgets")
@post
op create(@body widget: Widget): Widget;
```

		return fmt.Errorf("unmarshalling type %T: %s", w, err.Error())

		}

		}

func unpopulateStringArray[T ~string](data json.RawMessage, fn string, v *[]T, d string) error {
	if data == nil || string(data) == "null" {
		return nil
	}
	var encodedValue string
	if err := json.Unmarshal(data, &encodedValue); err != nil {
		return fmt.Errorf("struct field %s: %s", fn, err.Error())
	}
	if encodedValue == "" {
		*v = []T{}
		return nil
	}
	values := strings.Split(encodedValue, d)
	result := make([]T, len(values))
	for i := range values {
		result[i] = T(values[i])
	}
	*v = result
	return nil
}

func populateStringArray[T ~string](m map[string]any, k string, v []T, d string) {
	if azcore.IsNullValue(v) {
		m[k] = nil
	} else if v != nil {
		encodedValue := make([]string, len(v))
		for i := range v {
			encodedValue[i] = string(v[i])
		}
		m[k] = strings.Join(encodedValue, d)
	}
}
