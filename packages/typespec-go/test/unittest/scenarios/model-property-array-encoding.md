# Delimited string and enum array model properties use their encoded wire format

## TypeSpec

```tsp
@service(#{ title: "ArrayEncoding" })
namespace ArrayEncoding;

enum Color {
  Red: "red",
  Blue: "blue",
}

union ExtensibleColor {
  string,
  Red: "red",
  Blue: "blue",
}

model Widget {
  @encode(TypeSpec.ArrayEncoding.commaDelimited)
  names: string[];

  @encode(TypeSpec.ArrayEncoding.pipeDelimited)
  optionalNames?: string[];

  @encode(TypeSpec.ArrayEncoding.spaceDelimited)
  colors: Color[];

  @encode(TypeSpec.ArrayEncoding.newlineDelimited)
  extensibleColors: ExtensibleColor[];
}

@route("/widgets")
@post
op create(@body widget: Widget): Widget;
```
