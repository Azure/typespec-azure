# A long-running operation with LRO status metadata and an operation ID gets a poller wrapper

## TypeSpec

```tsp
@service(#{ title: "LRO metadata" })
namespace Lro;

model JobStatus {
  @lroOperationId
  id: string;

  status: global.Azure.Core.Foundations.OperationState;

  @lroStatusMetadata
  percentComplete?: int32;

  result?: JobResult;
}

model JobResult {
  name: string;
}

@route("/jobs/operations/{operationId}")
@get
op getJobStatus(
  @lroOperationId
  @path
  operationId: string,
): JobStatus;

@route("/jobs:start")
@post
@global.Azure.Core.pollingOperation(getJobStatus)
op startJob(): global.Azure.Core.Foundations.AcceptedResponse &
  global.Azure.Core.Foundations.LongRunningStatusLocation<JobStatus>;
```
