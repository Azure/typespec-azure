# A renamed XML root preserves the nested property name in clients and fake servers

## TypeSpec

```tsp
@service(#{ title: "XML naming" })
namespace XmlNaming;

alias XmlContent = {
  @header("Content-Type")
  contentType: "application/xml";
};

@Xml.name("XmlAuthor")
model Author {
  name: string;
}

@Xml.name("XmlLibrary")
model Library {
  @Xml.name("author")
  author: Author;
}

@route("/library")
interface LibraryOperations {
  @get
  get(): XmlContent & {
    @body
    body: Library;
  };

  @put
  put(@header("Content-Type") contentType: "application/xml", @body body: Library): XmlContent & {
    @body
    body: Library;
  };
}
```
