```yaml
generate-samples: true
```

## TypeSpec

```tsp
@armProviderNamespace
namespace Microsoft.Test;

model ConfigItem {
  value: string;
}

model ModelConfig extends Record<ConfigItem> {}
model StringConfig extends Record<string> {}
model AnyConfig extends Record<unknown> {}

model SettingsProperties {
  modelConfig: ModelConfig;
  stringConfig: StringConfig;
  anyConfig: AnyConfig;
}

model Settings is TrackedResource<SettingsProperties> {
  @pattern("^[A-Za-z0-9]+$")
  @path
  @key("settingsName")
  @segment("settings")
  name: string;
}

@armResourceOperations
interface SettingsOps {
  put is ArmResourceCreateOrReplaceSync<Settings>;
}
```
