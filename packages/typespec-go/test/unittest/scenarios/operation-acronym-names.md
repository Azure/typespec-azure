# Public LRO and pager types preserve operation acronym casing

## TypeSpec

```tsp
@service
namespace Lro;

model WidgetOptions {
  name: string;
}

model Widget {
  name: string;
}

model WidgetStatus is global.Azure.Core.Foundations.OperationStatus<Widget>;
@@visibility(global.Azure.Core.Foundations.OperationStatus.id, Lifecycle.Read);

@route("/widgets:prepare")
@clientName("BMSPrepareDataMove")
op prepare is global.Azure.Core.LongRunningRpcOperation<
  {
    @bodyRoot body: WidgetOptions;
  },
  WidgetStatus,
  Widget
>;

@route("/widgets:trigger")
@clientName("BMSTriggerDataMove")
op trigger is global.Azure.Core.LongRunningRpcOperation<
  {
    @bodyRoot body: WidgetOptions;
  },
  WidgetStatus,
  Widget
>;

model WidgetList {
  @pageItems
  values: Widget[];
}

@route("/widgets")
@get
@list
@clientName("BMSList")
op list(): WidgetList;
```
