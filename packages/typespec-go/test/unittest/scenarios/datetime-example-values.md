# Example datetime values are emitted as readable UTC time.Date literals, not time.Parse of the wire format

Generated samples construct datetimes with `time.Date(..., time.UTC)` (or
`time.Unix` for epoch timestamps) rather than leaking the wire format into a
`time.Parse` call. A `utcDateTime` value carrying a non-UTC offset is folded to
UTC and RFC7231 is handled with the RFC1123 grammar, while an `offsetDateTime`
value preserves its authored offset via `time.FixedZone`.

```yaml
generate-samples: true
```

## TypeSpec

```tsp
@armProviderNamespace
namespace Microsoft.Test;

model TimesProperties {
  rfc3339: utcDateTime;

  @encode(DateTimeKnownEncoding.rfc7231)
  rfc7231: utcDateTime;

  @encode(DateTimeKnownEncoding.unixTimestamp, int64)
  unix: utcDateTime;

  plainDate: plainDate;
  plainTime: plainTime;
  offset: offsetDateTime;
  offsetZulu: offsetDateTime;
}

model Times is TrackedResource<TimesProperties> {
  @pattern("^[A-Za-z0-9]+$")
  @path
  @key("timesName")
  @segment("times")
  name: string;
}

@armResourceOperations
interface TimesOps {
  put is ArmResourceCreateOrReplaceSync<Times>;
}
```
